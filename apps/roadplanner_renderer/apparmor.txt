#include <tunables/global>

# AppArmor profile for the Roadplanner renderer PoC.
#
# The app reads and writes exactly one tree - its exchange directory under
# /share - plus its own read-only program files. Everything the profile does
# not name is denied, which is what makes "minimal permissions" an enforced
# property rather than a claim in a README.
profile roadplanner_renderer_poc flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Container basics.
  network,
  capability,
  file,
  umount,

  # Nothing here needs to become another user or load a policy.
  deny @{PROC}/* w,
  deny /sys/fs/cgroup/** wklx,

  /run.sh r,
  /bin/** rix,
  /usr/bin/** rix,
  /usr/lib/** mr,

  # The worker and its (empty) dependency tree, read-only.
  /opt/roadplanner-renderer/** r,

  # The one writable area: the shared exchange directory.
  /share/roadplanner-renderer/** rwk,
}
